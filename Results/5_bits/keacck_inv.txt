ANF is given by : 
y0 = x0 ^ x2 ^ x1x2 ^ x4 ^ x1x4 ^ x3x4 ^ x1x3x4
y1 = x0 ^ x1 ^ x0x2 ^ x3 ^ x2x3 ^ x0x4 ^ x0x2x4
y2 = x1 ^ x0x1 ^ x2 ^ x1x3 ^ x0x1x3 ^ x4 ^ x3x4
y3 = x0 ^ x2 ^ x1x2 ^ x3 ^ x0x4 ^ x2x4 ^ x1x2x4
y4 = x1 ^ x0x1 ^ x3 ^ x0x3 ^ x2x3 ^ x0x2x3 ^ x4

The given s-box can be implemented by :

uint32_t Sbox(uint32_t X, uint8_t size) {
	uint32_t x[size];
	for(uint8_t b=0; b<size; b++)  {
		x[b] = X&1ul;
		X >>= 1;
	}
	uint32_t y[size];
	uint32_t l5 = x[1] ^ x[3];
	uint32_t l0 = x[4] ^ l5;
	uint32_t l4 = x[0] ^ l5;
	uint32_t l6 = x[2] ^ x[4];
	uint32_t l1 = x[0] ^ l6;
	uint32_t l2 = x[1] ^ l6;
	uint32_t l7 = x[4] ^ l1;
	uint32_t l3 = x[3] ^ l7;
	uint32_t q0 = x[3] & x[4];
	uint32_t q1 = x[2] & x[4];
	uint32_t q2 = x[2] & x[3];
	uint32_t q3 = x[1] & x[3];
	uint32_t q4 = q1 ^ q2;
	uint32_t q5 = q0 ^ q3;
	uint32_t ty4 = ((q2) ^ (l5)) & (x[0]) ^ (q2);
	uint32_t ty0 = ((q0) ^ (l6)) & (x[1]) ^ (q0);
	uint32_t ty2 = ((q3) ^ x[1]) & (x[0]) ^ (q5);
	uint32_t ty3 = ty4 ^ ((q2) ^ (l0)) & ((q1) ^ (l7)) ^ (q4);
	uint32_t ty1 = ((q1) ^ (l6)) & (x[0]) ^ (q2);
	y[4] = ty4 ^ l0;
	y[0] = ty0 ^ l1;
	y[2] = ty2 ^ l2;
	y[3] = ty3 ^ l3;
	y[1] = ty1 ^ l4;
	uint32_t Y = 0;
	for(uint8_t b=0; b<size; b++) {
		Y ^= ((Y>>b) ^ y[b]) << b;
	}
	return Y;
}
